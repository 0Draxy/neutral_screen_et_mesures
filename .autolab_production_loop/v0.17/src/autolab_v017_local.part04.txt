void TryEntry()
{{
   ManagePosition();
   if(HasPosition()) return;
   int direction=0;
   double atrValue=0.0;
   if(!Signal(direction,atrValue)) return;

   MqlTick tick;
   if(!SymbolInfoTick(_Symbol,tick) || tick.ask<=0.0 || tick.bid<=0.0) return;
   double spread=tick.ask-tick.bid;
   if(spread<=0.0 || spread>atrValue*MaxSpreadATR) return;

   double point=SymbolInfoDouble(_Symbol,SYMBOL_POINT);
   int digits=(int)SymbolInfoInteger(_Symbol,SYMBOL_DIGITS);
   long stopsLevel=SymbolInfoInteger(_Symbol,SYMBOL_TRADE_STOPS_LEVEL);
   double minDist=(double)(stopsLevel+2)*point;
   double stopDist=MathMax(atrValue*StopATR,minDist);
   if(stopDist<=0.0) return;

   ENUM_ORDER_TYPE type=(direction>0 ? ORDER_TYPE_BUY : ORDER_TYPE_SELL);
   double entry=(direction>0 ? tick.ask : tick.bid);
   double sl=(direction>0 ? entry-stopDist : entry+stopDist);
   double tp=(direction>0 ? entry+stopDist*RewardRisk : entry-stopDist*RewardRisk);
   sl=NormalizeDouble(sl,digits);
   tp=NormalizeDouble(tp,digits);

   double vol=RiskBasedVolume(type,entry,sl);
   if(vol<=0.0) return;

   bool ok=false;
   if(direction>0)
      ok=trade.Buy(vol,_Symbol,0.0,sl,tp,"AutoLabV017");
   else
      ok=trade.Sell(vol,_Symbol,0.0,sl,tp,"AutoLabV017");

   if(!ok)
      Print("Entry failed ",trade.ResultRetcode()," ",trade.ResultRetcodeDescription());
}}
