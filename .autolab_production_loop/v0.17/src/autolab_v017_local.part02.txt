def mutate_gene(parent, mut_index):
    g=dict(parent); fam=g["family"]; mi=int(mut_index)
    if mi==1:
        g["level1"]=round(b.clamp(float(g["level1"])+1.0,24.0,34.0),1)
        g["level2"]=round(b.clamp(float(g["level2"])-0.05,0.80,1.40),2)

        g["flat_atr"]=round(b.clamp(float(g["flat_atr"])-0.10,0.80,1.50),2)
    elif mi==3:
        g["rr"]=round(b.clamp(float(g["rr"])+0.10,1.00,1.50),2)
        g["max_hold"]=int(b.clamp(int(g["max_hold"])+2,6,14))
    else:
        if fam in ("reversion_slope_guard","reversion_session_guard","reversion_combo_guard"):
            g["slope_max"]=round(b.clamp(float(g["slope_max"])+0.15,0.20,1.75),2)

            g["vol_max"]=round(b.clamp(float(g["vol_max"])+0.10,1.10,1.90),2)
