        f"ResultId={result_id}\nNext=STOP - AUCUNE v0.18 AUTOMATIQUE\n"
        "Holdout2025_2026=NON OUVERT EN v0.17\n\n"+
        "\n".join(" - "+a for _,a in unique)+"\n",encoding="utf-8")
    unique.insert(0,(b.HANDOFF_INDEX,"LISEZ_MOI_A_ENVOYER_CHATGPT.txt"))
    tmp=b.HANDOFF_ZIP.with_suffix(".tmp")
    try: tmp.unlink()
    except FileNotFoundError: pass
    with zipfile.ZipFile(tmp,"w",zipfile.ZIP_DEFLATED,compresslevel=6) as z:
        for p,a in unique:
            z.write(p,a)
    tmp.replace(b.HANDOFF_ZIP)
    b.log("PAQUET CHATGPT PRET: "+str(b.HANDOFF_ZIP))
    return b.HANDOFF_ZIP
b.prepare_handoff=prepare_handoff

def autoloop_finish(c,reason=""):
    if not bool(c.get("autoloop_enabled",True)):
        b.log("AUTOLOOP desactive."); return
    if not b.HANDOFF_ZIP.exists():
        b.log("AUTOLOOP impossible: ZIP resultat absent."); return
    result_id=b.ensure_loop_id()
    handoff_hash=b.sha256(b.HANDOFF_ZIP)
    b.log("===== AUTOLOOP V0.17 RESULTAT -> CHATGPT -> STOP =====")
    b.log("ResultId="+result_id)
    b.log("HandoffSHA256="+handoff_hash)
    if not b.AUTOLOOP_UPLOAD.exists():
        b.log("Uploader robuste absent."); return
    if not b.AUTOLOOP_UPLOAD_MARKER.exists():
        msg=("AUTOLAB PRODUCTION LOOP : resultat v0.17 automatique. "
             "Analyse le ZIP v0.17. La boucle demandee est terminee : STOP. "
             "Ne publie pas v0.18.")
